cmake_minimum_required(VERSION 3.20)
project(emulard_probes LANGUAGES C CXX)
add_executable(abi_smoke abi_smoke.c)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Use the MSVC or clang-cl Windows toolchain.")
  endif()
  target_compile_options(abi_smoke PRIVATE /GS-)
  target_link_options(abi_smoke PRIVATE /ENTRY:mainCRTStartup /SUBSYSTEM:CONSOLE /NODEFAULTLIB)
  add_executable(graphics_caps graphics_caps.cpp)
  target_compile_features(graphics_caps PRIVATE cxx_std_17)
  target_link_libraries(graphics_caps PRIVATE d3d12 dxgi)
endif()
enable_testing()
add_test(NAME abi_smoke COMMAND abi_smoke)
